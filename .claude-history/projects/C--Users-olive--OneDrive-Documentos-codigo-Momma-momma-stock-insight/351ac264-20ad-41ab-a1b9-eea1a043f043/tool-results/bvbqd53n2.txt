  if (!d) return '';
  try {
    return new Date(d).toLocaleDateString('pt-BR', {
      day: '2-digit', month: '2-digit', year: '2-digit',
    });
  } catch {
    return d;
  }
};

export function CmvAtrelarNFDialog({ open, cmvItemId, embalagemNome, currentNfItemId, onClose, onSaved }: CmvAtrelarNFDialogProps) {

  const [items, setItems] = useState<NfItemData[]>([]);

  const [saving, setSaving] = useState(false);

      setStep('pickNF');
      setPickedNF(null);
      setBusca('');
      loadNFs();

    setLoading(true);
    const { data, error } = await supabase
      .from('notas_fiscais')
      .select('id, fornecedor, numero_nota, valor_total, data_nota')
      .order('data_nota', { ascending: false })
      .limit(80);
      
    if (!error && data) {
      setNfs(data);
    }
    setLoading(false);
  };

  const loadItems = async (notaId: number) => {
    setLoading(true);
    const { data, error } = await supabase
      .from('itens_nota_fiscal')
      .select('id, nota_id, nome_produto, quantidade, valor_unitario, subtotal, insumo_id')
      .eq('nota_id', notaId)
      .order('nome_produto');

    if (!error && data) {
      setItems(data);
    }
    setLoading(false);

    setBusca('');

        preco_unitario_esboco: null
      })
      .eq('id', cmvItemId);

      toast.error(`Erro ao atrelar: ${error.message}`);

      toast.success('Nota atrelada com sucesso!');

      onClose();

              <>
                <Button

                >
                  Outra NF

              </>
            )}

              Atrelar NF para {embalagemNome || 'Embalagem'}
            </span>

      toast.success('Excluído!');
      await onRefetch();
      onClose();

import { useState } from 'react';
import { useReactFlow } from '@xyflow/react';

import { cn } from '@/shared/lib/utils';

interface CmvBottomToolbarProps {
  onSavePosition?: (nodeId: string, x: number, y: number) => Promise<void>;
}

export function CmvBottomToolbar({ onSavePosition }: CmvBottomToolbarProps) {
  const { zoomIn, zoomOut, fitView, setCenter, getNodes, getEdges, setNodes } = useReactFlow();
  const [isLocked, setIsLocked] = useState(false);
  const [isOrganizing, setIsOrganizing] = useState(false);

  const handleLayout = () => {
    setIsOrganizing(true);
    const nodes = getNodes();
    const edges = getEdges();
    const laid = layoutGraph(nodes, edges, 'LR');
    
    setNodes(laid);
    
    if (onSavePosition) {
      laid.forEach(n => {
        onSavePosition(n.id, n.position.x, n.position.y);
      });
    }

    setTimeout(() => {
      fitView({ padding: 0.12, duration: 600 });
      setIsOrganizing(false);
    }, 100);
  };

  const toggleLock = () => {
    const newState = !isLocked;
    setIsLocked(newState);
    setNodes(nds => nds.map(n => ({ ...n, draggable: !newState })));
  };

  const ToolbarBtn = ({ icon, title, onClick, active }: { icon: React.ReactNode, title: string, onClick: () => void, active?: boolean }) => (
    <button
      title={title}
      onClick={onClick}
      className={cn(
        "h-9 w-9 rounded-lg flex items-center justify-center transition-all",
        active 
          ? "bg-[#57715B] text-white shadow-md hover:bg-[#465b49]" 
          : "text-slate-600 dark:text-slate-300 hover:bg-[#57715B]/10 hover:text-[#57715B]"
      )}
    >
      {icon}
    </button>
  );

  const Divider = () => <div className="w-px h-6 bg-[#57715B]/10 mx-0.5" />;

  return (
    <div className="absolute bottom-4 left-1/2 -translate-x-1/2 z-20 flex items-center gap-1 rounded-2xl bg-white/95 dark:bg-slate-900/95 backdrop-blur-md border border-white/40 dark:border-white/10 shadow-2xl px-2 py-1.5 animate-in fade-in slide-in-from-bottom-4 duration-300">

      <Divider />
      
      <ToolbarBtn icon={<Maximize2 size={16} />} title="Ajustar à tela" onClick={() => fitView({ duration: 600 })} />

    }

    const addedNfNodeIds = new Set<number>();

    items.forEach((item, idx) => {
      const itemId = `item_${item.id}`;
      const label = item.produtoMaster?.nome ?? item.insumo?.nome ?? item.observacao ?? 'Embalagem';

      newNodes.push({
        id: itemId,
        type: 'cmvItem',
        data: {
          tipo: item.tipo as TipoItem,
          quantidade: item.quantidade,
          unidade: item.unidade,
          custo: item.custo_calculado,
          label,
          kind: 'cmvItem',
          _rawItemId: item.id,
        } as Record<string, unknown>,
        position: positions.get(itemId) ?? { x: 400, y: 100 + idx * 160 },
      });

      newEdges.push({
        id: `e-${itemId}-${rootId}`,
        source: itemId,
        target: rootId,
        style: { stroke: '#94a3b8', strokeWidth: 1.5 },
      });

      if (item.nf_item_id && item.nfItem && item.nfItem.nota_id) {
        const notaId = item.nfItem.nota_id;
        const nfNodeKey = `nf_nota_${notaId}`;
        
        if (!addedNfNodeIds.has(notaId)) {
          addedNfNodeIds.add(notaId);
          newNodes.push({
            id: nfNodeKey,
            type: 'nf',
            data: {
              fornecedor: item.nfItem.fornecedor ?? 'Fornecedor',
              numero: item.nfItem.numero_nota,
              valor: null,
              kind: 'nf',
            } as Record<string, unknown>,
            position: positions.get(nfNodeKey) ?? { x: 50, y: 50 + idx * 160 },
          });
        }

        newEdges.push({
          id: `e-${nfNodeKey}-${itemId}`,
          source: nfNodeKey,
          target: itemId,
          style: { stroke: '#06b6d4', strokeWidth: 1.5, strokeDasharray: '4 2' },
        });
      }
    });

    const hasAllPositions = newNodes.every((n) => positions.has(n.id));
    if (!hasAllPositions && newNodes.length > 0) {
      const laid = layoutGraph(newNodes, newEdges, 'LR');
      setNodes(laid);
    } else {
      setNodes(newNodes);
    }
    setEdges(newEdges);

    setTimeout(() => fitView({ padding: 0.12, duration: 400 }), 80);
  }, [cmv, fichaData, items, positions, fitView, setNodes, setEdges]);

  useEffect(() => {
    if (!loading) buildGraph();
  }, [loading, buildGraph]);

  const handleNodeDragStop: NodeDragHandler = useCallback((_evt, node) => {
    savePosition(node.id, node.position.x, node.position.y);
  }, [savePosition]);

  const handleNodeClick: NodeMouseHandler = useCallback((_evt, node) => {
    setSelectedNodeId(node.id);
  }, []);

  const handlePaneClick = useCallback(() => {
    setSelectedNodeId(null);
  }, []);

  const handleAddEmbalagem = useCallback(async (embId: number) => {
    if (!cmvId) return;
    const { error } = await supabase.from('cmv_itens').insert({
      cmv_id: cmvId, tipo: 'embalagem', produto_master_id: embId,
      quantidade: 1, unidade: 'un', ordem: items.length
    } as Record<string, unknown>);
    if (error) { toast.error(error.message); return; }
    toast.success('Embalagem adicionada');
    await refetch();
  }, [cmvId, items.length, refetch]);

  const handleAddAvulso = useCallback(async (nome: string, valor: number) => {
    if (!cmvId) return;
    const { error } = await supabase.from('cmv_itens').insert({
      cmv_id: cmvId, tipo: 'outro', observacao: nome,
      preco_unitario_esboco: valor, quantidade: 1, unidade: 'un', ordem: items.length
    } as Record<string, unknown>);
    if (error) { toast.error(error.message); return; }
    toast.success('Custo avulso adicionado');
    await refetch();
  }, [cmvId, items.length, refetch]);

  if (!cmvId || (!cmv && !loading)) {
    return (
      <div className="h-[100dvh] bg-white dark:bg-[#0A0A0A] flex flex-col items-center justify-center text-muted-foreground">
        <Calculator className="h-12 w-12 mb-4 text-[#57715B]" />
        <p className="text-sm mb-4">Nenhum CMV selecionado</p>
        <Button variant="outline" className="rounded-xl border-[#57715B]/20 bg-white/60 hover:bg-[#57715B]/10 text-[#57715B]" onClick={() => navigate('/financeiro/cmv')}>
          <ArrowLeft className="mr-2 h-4 w-4" /> Voltar ao CMV
        </Button>
      </div>
    );
  }

  if (loading) {
    return <div className="h-[100dvh] bg-white dark:bg-[#0A0A0A] flex items-center justify-center"><Loader2 className="h-8 w-8 animate-spin text-[#57715B]" /></div>;
  }

  return (
    <div className="fixed inset-0 bg-white dark:bg-[#0A0A0A] overflow-hidden">
      <div className="fixed top-[-10%] left-[-10%] w-[500px] h-[500px] bg-[#57715B]/5 blur-[80px] rounded-full pointer-events-none" />
      <div className="fixed top-[20%] right-[-5%] w-[400px] h-[400px] bg-[#C5A065]/5 blur-[80px] rounded-full pointer-events-none" />

      <ReactFlow
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        nodeTypes={nodeTypes}
        onNodeDragStop={handleNodeDragStop}
        onNodeClick={handleNodeClick}
        onPaneClick={handlePaneClick}
        fitView
        proOptions={{ hideAttribution: true }}
        minZoom={0.2}
        maxZoom={2}
      >
        <Background variant={BackgroundVariant.Dots} color="#57715B" gap={24} className="opacity-20" />
        <MiniMap className="!bg-white/90 !backdrop-blur !rounded-xl !shadow-md !border !border-[#57715B]/10" nodeColor={() => '#57715B'} maskColor="rgba(0,0,0,0.4)" />

        <Panel position="top-left">
          <div className="flex items-center gap-2 bg-white/90 dark:bg-[#1A1A1A]/90 border border-[#57715B]/10 backdrop-blur-xl rounded-xl px-3 py-2 shadow-xl">
            <Button variant="ghost" size="icon" className="h-9 w-9 text-muted-foreground hover:bg-[#57715B]/10 hover:text-[#57715B]" onClick={() => navigate('/financeiro/cmv')}>
              <ArrowLeft size={16} />
            </Button>
            <div className="w-px h-5 bg-[#57715B]/15" />
            <div className="flex items-center gap-2 px-1">
              <Calculator size={14} className="text-[#57715B]" />
              <span className="text-sm font-semibold text-[#2D362E] dark:text-[#E6DCCF] truncate max-w-[180px]">{cmv?.nome}</span>
              <span className={`text-[10px] px-2 py-0.5 rounded-full font-bold uppercase tracking-wide ${STATUS_COLORS[cmv?.status || 'rascunho'] ?? STATUS_COLORS.rascunho}`}>
                {cmv?.status}
              </span>
            </div>
          </div>
        </Panel>

        <Panel position="top-right">
          <div className="flex flex-col gap-1.5 bg-white/90 dark:bg-[#1A1A1A]/90 border border-[#57715B]/10 backdrop-blur-xl rounded-xl p-3 shadow-lg text-[10px] text-muted-foreground">
            <p className="text-[9px] font-bold uppercase tracking-widest mb-1 text-[#57715B]">Legenda</p>
            <div className="flex items-center gap-1.5"><Calculator size={10} className="text-[#57715B]" /><span>CMV Central</span></div>
            <div className="flex items-center gap-1.5"><ChefHat size={10} className="text-amber-500" /><span>Ficha Técnica</span></div>
            <div className="flex items-center gap-1.5"><Package size={10} className="text-blue-500" /><span>Embalagem</span></div>
            <div className="flex items-center gap-1.5"><DollarSign size={10} className="text-orange-500" /><span>Custo avulso</span></div>
            <div className="flex items-center gap-1.5"><Receipt size={10} className="text-cyan-500" /><span>Nota Fiscal</span></div>
          </div>
        </Panel>

        <Panel position="bottom-left">
          <div className="bg-white/90 dark:bg-[#1A1A1A]/90 backdrop-blur-xl border border-[#57715B]/10 rounded-2xl p-4 shadow-xl min-w-[200px] max-w-[calc(100vw-1rem)]">
            <p className="text-[9px] font-bold uppercase tracking-widest text-muted-foreground mb-3">Breakdown de Custo</p>
            <div className="space-y-1.5">
              <div className="flex justify-between items-baseline">
                <span className="text-xs text-muted-foreground">Ficha base (unitária)</span>
                <span className="text-sm tabular-nums text-[#2D362E] dark:text-[#E6DCCF]">{fmt(custoFichaUnitario)}</span>
              </div>
              <div className="flex justify-between items-baseline">
                <span className="text-xs text-muted-foreground">Embalagens</span>
                <span className="text-sm tabular-nums text-[#2D362E] dark:text-[#E6DCCF]">{fmt(custoEmbalagens)}</span>
              </div>
              <div className="flex justify-between items-baseline">
                <span className="text-xs text-muted-foreground">Custos avulsos</span>
                <span className="text-sm tabular-nums text-[#2D362E] dark:text-[#E6DCCF]">{fmt(custoAvulsos)}</span>
              </div>
              <div className="border-t border-[#57715B]/10 my-2" />
              <div className="flex justify-between items-baseline">
                <span className="text-sm font-bold text-muted-foreground">= Custo total</span>
                <span className="text-sm font-bold text-[#57715B] tabular-nums">{fmt(custoTotal)}</span>
              </div>
            </div>
          </div>
        </Panel>
      </ReactFlow>

      <CmvNodePalette embalagens={embalagens} onAddEmbalagem={handleAddEmbalagem} onAddAvulso={handleAddAvulso} />

      <CmvDetailPanel
        selectedNodeId={selectedNodeId}
        cmv={cmv}
        fichaData={fichaData}
        items={items}
        custoFichaUnitario={custoFichaUnitario}
        custoEmbalagens={custoEmbalagens}
        custoAvulsos={custoAvulsos}
        custoTotal={custoTotal}
        nfEmbalagemItems={nfEmbalagemItems}
        onClose={() => setSelectedNodeId(null)}
        onRefetch={refetch}
      />

      <CmvBottomToolbar onSavePosition={savePosition} />
    </div>
  );
}

export default function CmvConfiguratorPage() {
  return (
    <ReactFlowProvider>
      <CmvConfiguratorInner />
    </ReactFlowProvider>
  );
}
