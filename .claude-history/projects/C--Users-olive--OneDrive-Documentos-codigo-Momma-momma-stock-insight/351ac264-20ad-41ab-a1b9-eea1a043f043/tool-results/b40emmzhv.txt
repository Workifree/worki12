const formatBRL = (value: number | null | undefined) => {
  if (value === null || typeof value === 'undefined') return '';
  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value);
};

function CmvConfigurator() {
  const navigate = useNavigate();
  const [searchParams] = useSearchParams();
  const cmvId = Number(searchParams.get('id'));
  const queryClient = useQueryClient();
  const {
    loading, cmv, fichaData, items, insumos, embalagens, nfEmbalagemItems, positions, savePosition, refetch
  } = useCmvGraph(cmvId);

  const [nodes, setNodes, onNodesChange] = useNodesState([]);
  const [edges, setEdges, onEdgesChange] = useEdgesState([]);
  const { setViewport, getViewport } = useReactFlow();

  const [isAddOpen, setIsAddOpen] = useState(false);
  const [isAddSaving, setIsAddSaving] = useState(false);
  const [newTipo, setNewTipo] = useState<'embalagem' | 'outro'>('embalagem');
  const [newEmbalagemId, setNewEmbalagemId] = useState('');
  const [newEmbalagemOpen, setNewEmbalagemOpen] = useState(false);
  const [newEmbalagemSearch, setNewEmbalagemSearch] = useState('');
  const [newFonteCusto, setNewFonteCusto] = useState<'nf' | 'fixo'>('nf');
  const [newNfItemId, setNewNfItemId] = useState('');
  const [newNfOpen, setNewNfOpen] = useState(false);
  const [newNfSearch, setNewNfSearch] = useState('');
  const [newUnidadesPorEmbalagem, setNewUnidadesPorEmbalagem] = useState('');
  const [newPrecoFixo, setNewPrecoFixo] = useState('');
  const [newQty, setNewQty] = useState('1');
  const [newUnit, setNewUnit] = useState('');
  const [newObs, setNewObs] = useState('');
  const [newPrecoOutro, setNewPrecoOutro] = useState('');

  const [selectedItemData, setSelectedItemData] = useState<RawCmvItem | null>(null);
  const [isEditSaving, setIsEditSaving] = useState(false);
  const [editQty, setEditQty] = useState('');
  const [editUnit, setEditUnit] = useState('');
  const [editObs, setEditObs] = useState('');
  const [editPreco, setEditPreco] = useState('');
  const [editFonteCusto, setEditFonteCusto] = useState<'nf' | 'fixo'>('fixo');
  const [editNfItemId, setEditNfItemId] = useState('');
  const [editNfOpen, setEditNfOpen] = useState(false);
  const [editNfSearch, setEditNfSearch] = useState('');
  const [editUnidadesPorEmbalagem, setEditUnidadesPorEmbalagem] = useState('');


  const buildGraph = useCallback(() => {
    if (!cmv) return;

    const newNodes: Node[] = [];
    const newEdges: Edge[] = [];
    const rootId = cmvNodeId(cmv.id);

    newNodes.push({
      id: rootId, type: 'cmv',
      data: { label: cmv.nome, custo: cmv.custo_calculado, kind: 'cmv' },
      position: positions.get(rootId) ?? { x: 400, y: 50 },
    });

    if (fichaData) {
      const fichaId = fichaNodeId(fichaData.id);
      const custoUnitario = (fichaData.custo_calculado ?? 0) / Math.max(1, fichaData.rendimento_quantidade);
      newNodes.push({
        id: fichaId, type: 'ficha',
        data: {
          label: fichaData.nome,
          subLabel: `${fichaData.rendimento_quantidade} ${fichaData.rendimento_unidade}`,
          custo: custoUnitario,
          kind: 'ficha',
        },
        position: positions.get(fichaId) ?? { x: 400, y: 250 },
      });
      newEdges.push({ id: `e-${rootId}-${fichaId}`, source: rootId, target: fichaId, style: { stroke: '#a1a1aa' } });
    }

    items.forEach((item, idx) => {
      const itemNodeId = `item_${item.id}`;
      const label = item.produtoMaster?.nome ?? item.insumo?.nome ?? item.observacao ?? `Item ${idx + 1}`;
      const subLabel = `${item.quantidade} ${item.unidade}`;
      newNodes.push({
        id: itemNodeId, type: 'item',
        data: { label, subLabel, custo: item.custo_calculado, kind: 'item' },
        position: positions.get(itemNodeId) ?? { x: 200 + idx * 250, y: 450 },
      });
      newEdges.push({ id: `e-${rootId}-${itemNodeId}`, source: rootId, target: itemNodeId, style: { stroke: '#a1a1aa' } });

      if (item.insumo_id && item.insumo) {
        const insumoId = insumoNodeId(item.insumo_id);
        newNodes.push({
          id: insumoId, type: 'insumo',
          data: {
            label: item.insumo.nome,
            subLabel: `${item.insumo.unidade_compra} @ ${formatBRL(item.insumo.preco_custo)}`,
            custo: item.insumo.preco_custo,
            kind: 'insumo',
          },
          position: positions.get(insumoId) ?? { x: 200 + idx * 250, y: 650 },
        });
        newEdges.push({ id: `e-${itemNodeId}-${insumoId}`, source: itemNodeId, target: insumoId, style: { stroke: '#fca5a5' } });
      }
    });

    const addedNfNodeIds = new Set<number>();
    items.forEach((item) => {
      if (item.nf_item_id && item.nfItem && !addedNfNodeIds.has(item.nf_item_id)) {
        addedNfNodeIds.add(item.nf_item_id);
        const nfId = nfNodeId(item.nf_item_id);
        newNodes.push({
          id: nfId, type: 'nf',
          data: {
            fornecedor: item.nfItem.fornecedor ?? 'Fornecedor',
            numero: item.nfItem.numero_nota,
            valor: item.nfItem.valor_unitario,
            kind: 'nf'
          } as Record<string, unknown>,
          position: positions.get(nfId) ?? { x: 50, y: 50 },
        });
      }
    });
    items.forEach((item) => {
      if (item.nf_item_id && item.nfItem) {
        const nfId = nfNodeId(item.nf_item_id);
        const itemNodeId = `item_${item.id}`;
        newEdges.push({ id: `e-${nfId}-${itemNodeId}`, source: nfId, target: itemNodeId, style: { stroke: '#22d3ee', strokeWidth: 1.5, strokeDasharray: '4 2' } });
      }
    });

    setNodes(newNodes);
    setEdges(newEdges);
  }, [cmv, fichaData, items, insumos, positions, setNodes, setEdges]);

  useEffect(() => {
    buildGraph();
  }, [buildGraph]);

  const handleNodeDragStop: NodeDragHandler = useCallback((_, node) => {
    savePosition(node.id, node.position);
  }, [savePosition]);

  const handleAutoLayout = useCallback(async () => {
    const { nodes: layoutedNodes, edges: layoutedEdges } = await layoutGraph(nodes, edges);
    layoutedNodes.forEach(node => savePosition(node.id, node.position));
    setNodes(layoutedNodes);
    setEdges(layoutedEdges);
    // Fit view after a short delay to allow for render
    setTimeout(() => {
        const { x, y, zoom } = getViewport();
        setViewport({ x: x+1, y, zoom }, { duration: 800 });
    }, 100);
  }, [nodes, edges, savePosition, setNodes, setEdges, getViewport, setViewport]);

  const resetAddDialog = () => {
    setNewTipo('embalagem');
    setNewEmbalagemId('');
    setNewEmbalagemSearch('');
    setNewFonteCusto('nf');
    setNewNfItemId('');
    setNewNfSearch('');
    setNewUnidadesPorEmbalagem('');
    setNewPrecoFixo('');
    setNewQty('1');
    setNewUnit('');
    setNewObs('');
    setNewPrecoOutro('');
  }

  const handleAddItem = async () => {
    if (!cmvId) return;
    setIsAddSaving(true);
    
    let insertPayload: Partial<RawCmvItem> | null = null;
    
    if (newTipo === 'embalagem') {
      if (!newEmbalagemId || !newQty || !newUnit) {
        toast.error("Embalagem, quantidade e unidade são obrigatórios.");
        setIsAddSaving(false);
        return;
      }
      insertPayload = {
        cmv_id: cmvId,
        tipo: 'embalagem',
        produto_master_id: parseInt(newEmbalagemId, 10),
        nf_item_id: newFonteCusto === 'nf' && newNfItemId ? parseInt(newNfItemId, 10) : null,
        unidades_por_embalagem: newFonteCusto === 'nf' && newUnidadesPorEmbalagem ? parseFloat(newUnidadesPorEmbalagem) : null,
        preco_unitario_esboco: newFonteCusto === 'fixo' && newPrecoFixo ? parseFloat(newPrecoFixo) : null,
        quantidade: parseFloat(newQty) || 1,
        unidade: newUnit,
        ordem: items.length,
      };
    } else { // tipo 'outro'
      if (!newObs.trim() || !newQty || !newUnit || !newPrecoOutro) {
        toast.error("Observação, preço, quantidade e unidade são obrigatórios.");
        setIsAddSaving(false);
        return;
      }
      insertPayload = {
        cmv_id: cmvId,
        tipo: 'outro',
        observacao: newObs.trim(),
        preco_unitario_esboco: newPrecoOutro ? parseFloat(newPrecoOutro) : null,
        quantidade: parseFloat(newQty) || 1,
        unidade: newUnit,
        ordem: items.length,
      };
    }

    if (insertPayload) {
      const { error } = await supabase.from('cmv_item').insert(insertPayload as any);
      if (error) {
        toast.error("Erro ao adicionar item.", { description: error.message });
      } else {
        toast.success("Item adicionado com sucesso!");
        await refetch();
        setIsAddOpen(false);
        resetAddDialog();
      }
    }
    setIsAddSaving(false);
  };

  const handleNodeClick = (_: React.MouseEvent, node: Node) => {
    const parsed = parseNodeId(node.id);
    if (parsed?.type === 'item') {
      const foundItem = items.find(i => i.id === parsed.id);
      if (foundItem) {
        setSelectedItemData(foundItem);
      }
    }
  };

  useEffect(() => {
    if (selectedItemData) {
      const found = items.find(i => i.id === selectedItemData.id);
      if (!found) {
        setSelectedItemData(null);
        return;
      }
      setEditQty(String(found.quantidade));
      setEditUnit(found.unidade);
      setEditObs(found.observacao ?? '');
      setEditPreco(found.preco_unitario_esboco !== null ? String(found.preco_unitario_esboco) : '');
      setEditFonteCusto(found.nf_item_id ? 'nf' : 'fixo');
      setEditNfItemId(found.nf_item_id ? String(found.nf_item_id) : '');
      setEditUnidadesPorEmbalagem(found.unidades_por_embalagem ? String(found.unidades_por_embalagem) : '');
    }
  }, [selectedItemData, items]);
  

  const handleSaveItem = async () => {
    if (!selectedItemData) return;
    setIsEditSaving(true);
    let updatePayload: Partial<RawCmvItem> | null = null;
    
    if (selectedItemData.tipo === 'embalagem') {
      updatePayload = {
        quantidade: parseFloat(editQty) || 1,
        unidade: editUnit,
        nf_item_id: editFonteCusto === 'nf' && editNfItemId ? parseInt(editNfItemId, 10) : null,
        unidades_por_embalagem: editFonteCusto === 'nf' && editUnidadesPorEmbalagem ? parseFloat(editUnidadesPorEmbalagem) : null,
        preco_unitario_esboco: editFonteCusto === 'fixo' && editPreco ? parseFloat(editPreco) : null,
      };
    } else { // tipo 'outro'
      updatePayload = {
        quantidade: parseFloat(editQty) || 1,
        unidade: editUnit,
        observacao: editObs.trim(),
        preco_unitario_esboco: editPreco ? parseFloat(editPreco) : null,
      };
    }

    const { error } = await supabase.from('cmv_item').update(updatePayload as any).eq('id', selectedItemData.id);
    if (error) {
      toast.error("Erro ao salvar alterações.", { description: error.message });
    } else {
      toast.success("Item atualizado com sucesso!");
      await refetch();
      setSelectedItemData(null);
    }
    setIsEditSaving(false);
  };
  
  const handleDeleteItem = async () => {
    if (!selectedItemData) return;
    setIsEditSaving(true);
    const { error } = await supabase.from('cmv_item').delete().eq('id', selectedItemData.id);
    if (error) {
      toast.error("Erro ao excluir item.", { description: error.message });
    } else {
      toast.success("Item excluído com sucesso!");
      await refetch();
      setSelectedItemData(null);
    }
    setIsEditSaving(false);
  };

  const selectedEmbalagem = useMemo(() => {
    if (!newEmbalagemId) return null;
    return embalagens.find(e => String(e.id) === newEmbalagemId);
  }, [newEmbalagemId, embalagens]);

  const nfCustoHint = useMemo(() => {
    if (newFonteCusto !== 'nf' || !newNfItemId || !newUnidadesPorEmbalagem) return null;
    const item = nfEmbalagemItems.find(i => String(i.id) === newNfItemId);
    const units = parseFloat(newUnidadesPorEmbalagem);
    if (!item || !item.valor_unitario || !units || units <= 0) return null;
    return item.valor_unitario / units;
  }, [newFonteCusto, newNfItemId, newUnidadesPorEmbalagem, nfEmbalagemItems]);

  const editNfCustoHint = useMemo(() => {
    if (editFonteCusto !== 'nf' || !editNfItemId || !editUnidadesPorEmbalagem) return null;
    const item = nfEmbalagemItems.find(i => String(i.id) === editNfItemId);
    const units = parseFloat(editUnidadesPorEmbalagem);
    if (!item || !item.valor_unitario || !units || units <= 0) return null;
    return item.valor_unitario / units;
  }, [editFonteCusto, editNfItemId, editUnidadesPorEmbalagem, nfEmbalagemItems]);

  const intelligentNfFilter = useMemo(() => {
    if (!selectedEmbalagem) return nfEmbalagemItems;
    const searchWords = selectedEmbalagem.nome.toLowerCase().split(/\s+/).filter(w => w.length > 2);
    return nfEmbalagemItems.filter(item => {
        if (!item.nome_produto) return false;
        const productName = item.nome_produto.toLowerCase();
        return searchWords.some(word => productName.includes(word));
    });
  }, [selectedEmbalagem, nfEmbalagemItems]);

  const filteredNfItems = useMemo(() => {
    const sourceList = newNfOpen ? intelligentNfFilter : nfEmbalagemItems;
    if (!newNfSearch) return sourceList;
    return sourceList.filter(item =>
        item.nome_produto?.toLowerCase().includes(newNfSearch.toLowerCase()) ||
        item.fornecedor?.toLowerCase().includes(newNfSearch.toLowerCase()) ||
        item.numero_nota?.toLowerCase().includes(newNfSearch.toLowerCase())
    );
  }, [newNfSearch, nfEmbalagemItems, newNfOpen, intelligentNfFilter]);

  const filteredEditNfItems = useMemo(() => {
    if (!editNfSearch) return nfEmbalagemItems;
    return nfEmbalagemItems.filter(item =>
        item.nome_produto?.toLowerCase().includes(editNfSearch.toLowerCase()) ||
        item.fornecedor?.toLowerCase().includes(editNfSearch.toLowerCase()) ||
        item.numero_nota?.toLowerCase().includes(editNfSearch.toLowerCase())
    );
  }, [editNfSearch, nfEmbalagemItems]);
  
  // Cost breakdown calculations
  const { custoFichaUnitario, custoEmbalagens, custoAvulsos, custoTotal } = useMemo(() => {
    const fichaUnitario = fichaData ? (fichaData.custo_calculado ?? 0) / Math.max(1, fichaData.rendimento_quantidade) : 0;
    const embalagensTotal = items.filter(i => i.tipo === 'embalagem').reduce((s,i) => s + (i.custo_calculado ?? 0), 0);
    const avulsosTotal = items.filter(i => i.tipo === 'outro').reduce((s,i) => s + (i.custo_calculado ?? 0), 0);
    const total = cmv?.custo_calculado ?? 0;
    return { custoFichaUnitario: fichaUnitario, custoEmbalagens: embalagensTotal, custoAvulsos: avulsosTotal, custoTotal: total };
  }, [items, cmv, fichaData]);

  if (loading) return <div className="h-screen w-full flex items-center justify-center bg-white dark:bg-[#0A0A0A]"><Loader2 className="h-8 w-8 animate-spin text-[#57715B]" /></div>;
  if (!cmvId || !cmv) return (
    <div className="h-screen w-full flex flex-col items-center justify-center bg-white dark:bg-[#0A0A0A] text-center">
      <h2 className="text-xl font-semibold text-gray-800 dark:text-gray-200">CMV não encontrado</h2>
      <p className="text-gray-500 dark:text-gray-400 mt-2">O Custo de Mercadoria Vendida que você está tentando acessar não existe ou foi removido.</p>
      <Button onClick={() => navigate('/financeiro/cmv')} className="mt-6 bg-[#57715B] hover:bg-[#465b49] text-white rounded-xl">Voltar para a lista</Button>
    </div>
  );

  return (
    <div className="w-full h-[calc(100vh_-_theme(spacing.14))] relative">
      <ReactFlow
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onNodeDragStop={handleNodeDragStop}
        onNodeClick={handleNodeClick}
        nodeTypes={nodeTypes}
        fitView
        className="bg-white dark:bg-[#0A0A0A]"
